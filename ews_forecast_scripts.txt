function show_hide(no)
{
	for(i=1;i<=10;i++)
	{
		if(i==no)
		{
			$("#panel"+i).removeClass("none");
			$("#panel"+i).addClass("showdiv");
		}
		else
		{
			$("#panel"+i).removeClass("showdiv");
			$("#panel"+i).addClass("none");
		}
	}
}
